# Wari Seva API — sample curl commands
# Base URL assumed: http://localhost:8000

# Health check
curl http://localhost:8000/health

# --- Users (/api/users) ---

# READ all users
curl "http://localhost:8000/api/users?action=READ"

# READ one user
curl "http://localhost:8000/api/users?action=READ&id=1"

# CREATE user
curl -X POST http://localhost:8000/api/users \
  -H "Content-Type: application/json" \
  -d '{"action":"CREATE","data":{"full_name":"Ram Patil","phone_number":"9876543210","role":"WALKER","dindi_id":1}}'

# UPDATE user (PATCH partial)
curl -X PATCH http://localhost:8000/api/users \
  -H "Content-Type: application/json" \
  -d '{"action":"UPDATE","id":1,"data":{"role":"VOLUNTEER"}}'

# DELETE user
curl -X DELETE "http://localhost:8000/api/users?action=DELETE&id=1"
